pragma solidity ^0.8.20;

contract TraceCoin {
    struct Evidence {
        address target;
        string pdfHash;
        uint256 timestamp;
        address submitter;
    }

    mapping(address => Evidence[]) private registry;
    mapping(address => mapping(string => bool)) private exists;

    event EvidenceStored(address indexed target, string pdfHash, uint256 timestamp, address indexed submitter);

    function storeEvidence(address _target, string memory _hash) public {
        require(_target != address(0), "target is zero");
        require(bytes(_hash).length == 64, "hash must be sha256 hex");
        require(!exists[_target][_hash], "duplicate evidence");

        registry[_target].push(Evidence({
            target: _target,
            pdfHash: _hash,
            timestamp: block.timestamp,
            submitter: msg.sender
        }));
        exists[_target][_hash] = true;

        emit EvidenceStored(_target, _hash, block.timestamp, msg.sender);
    }

    function getEvidences(address _target) public view returns (Evidence[] memory) {
        return registry[_target];
    }

    function hasEvidence(address _target, string memory _hash) public view returns (bool) {
        return exists[_target][_hash];
    }

    function getEvidenceCount(address _target) public view returns (uint256) {
        return registry[_target].length;
    }
}
